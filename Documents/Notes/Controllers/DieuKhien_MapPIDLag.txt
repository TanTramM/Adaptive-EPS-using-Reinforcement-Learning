[ T_{a,cmd} = H_{lag}(s)\,\big[u_{map} + u_{PID}\big], \quad H_{lag}(s) = \frac{s/z_1 + 1}{s/p_1 + 1}, \quad p_1 < z_1 ]

[ u_{map} = \mathrm{sgn}(T_s)\,M_K(v,|T_s|) ]
  (3) e_T = T_s - T_d,ref(v, a_y);   u_PID = Kp e_T + u_I + Kd d_f;   du_I/dt = Ki e_T + Kaw (sat(u_map + u_PID) - (u_map + u_PID)),   Kaw = Ki / (Kp + K)
[ e_T = T_s - T_{d,ref}(v,a_y), \quad u_{PID} = K_p e_T + u_I + K_d d_f, \quad \dot u_I = K_i e_T + K_{aw}\big(\mathrm{sat}(u_{map}+u_{PID}) - (u_{map}+u_{PID})\big), \quad K_{aw} = \frac{K_i}{K_p + K} ]
